cmake_minimum_required(VERSION 3.10)
project(qpiclocalsend C)
add_library(qpiclocalsend SHARED src/transport.c third_party/picohttpparser/picohttpparser.c)
target_include_directories(qpiclocalsend PRIVATE third_party/picohttpparser)
target_compile_features(qpiclocalsend PRIVATE c_std_99)
target_compile_options(qpiclocalsend PRIVATE -Os -fvisibility=hidden -Wall -Wextra -Wno-unused-parameter)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(qpiclocalsend PRIVATE ${JNI_INCLUDE_DIRS})
endif()
set_property(TARGET qpiclocalsend APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/jni/exports.map")
